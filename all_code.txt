const express = require('express');
const path = require('path');
const app = express();
const PORT = process.env.PORT || 3000;

app.use(express.json());
app.use(express.urlencoded({ extended: true }));
app.use(express.static(path.join(__dirname, 'public')));

let usersData = [
  { id: 'admin', pw: '1234', name: '최고관리자', role: 'admin', approved: true, points: 0, isSubscribed: true, hasSurvey: true }
];
let jobsData = [];
let surveysData = [];
let settlementsData = [];
let notificationsData = [];
let inquiriesData = [];
let invoicesData = []; 

// Auth & Users API
app.post('/api/signup', (req, res) => {
  const { id, pw, name, phone, role } = req.body;
  if (usersData.find(u => u.id === id)) return res.status(400).json({ success: false, message: '이미 존재하는 아이디입니다.' });
  const newUser = { id, pw, name, phone, role, approved: false, points: 0, isSubscribed: false, subRequested: false, hasSurvey: false };
  usersData.push(newUser);

  notificationsData.unshift({
    id: 'noti-' + Date.now(), targetUserId: id, title: '[회원가입 축하]',
    content: `${name}님, silverworks 회원가입을 진심으로 축하합니다!`,
    date: new Date().toLocaleString(), isRead: false
  });
  res.json({ success: true, user: newUser });
});

app.post('/api/login', (req, res) => {
  const { id, pw, role } = req.body;
  const user = usersData.find(u => u.id === id && u.role === role);
  if (!user) return res.status(400).json({ success: false, message: '등록되지 않았거나 회원 유형이 일치하지 않습니다.' });
  if (user.pw !== pw) return res.status(400).json({ success: false, message: '비밀번호가 일치하지 않습니다.' });
  if (!user.approved) return res.status(400).json({ success: false, message: '관리자의 가입 승인 대기 중입니다.' });
  res.json({ success: true, user });
});

app.get('/api/users', (req, res) => res.json({ success: true, users: usersData }));

app.post('/api/users/:id/approve', (req, res) => {
  const user = usersData.find(u => u.id === req.params.id);
  if (user) user.approved = true;
  res.json({ success: true });
});

app.post('/api/users/:id/update-points', (req, res) => {
  const { points } = req.body;
  const user = usersData.find(u => u.id === req.params.id);
  if (user) user.points = parseInt(points, 10) || 0;
  res.json({ success: true });
});

app.post('/api/users/:id/subscribe-request', (req, res) => {

app.post('/api/jobs', (req, res) => {
  const { employerId, storeName, title, category, date, time, pay, isPeak } = req.body;
  const employer = usersData.find(u => u.id === employerId);
  if (isPeak && (!employer || !employer.isSubscribed)) return res.status(403).json({ success: false, message: '상단 고정 공고는 구독 회원 사장님만 이용 가능합니다.' });

  const newJob = {
    id: 'job-' + Date.now(), employerId, storeName, title: isPeak ? `[상단 고정] ${title}` : title, category, date, time, pay: parseInt(pay, 10), isPeak: !!isPeak, status: '구직자 모집중', candidates: []
  };
  jobsData.unshift(newJob);
  res.json({ success: true, job: newJob });
});

app.put('/api/jobs/:id', (req, res) => {
  const job = jobsData.find(j => j.id === req.params.id);
  if (job) Object.assign(job, req.body);
  res.json({ success: true });
});

app.delete('/api/jobs/:id', (req, res) => {
  const jobIdx = jobsData.findIndex(j => j.id === req.params.id);
  if (jobIdx !== -1) {
    const job = jobsData[jobIdx];
    if (job.candidates) {
      job.candidates.forEach(c => {
        notificationsData.unshift({
          id: 'noti-' + Date.now(), targetUserId: c.seniorId, title: '[공고 삭제 알림]',
          content: `지원하신 '${job.title}' 공고가 삭제되어 내역이 정리되었습니다.`,
          date: new Date().toLocaleString(), isRead: false
        });
      });
    }
    jobsData.splice(jobIdx, 1);
  }
  res.json({ success: true });
});

app.post('/api/jobs/:id/apply', (req, res) => {
  const { seniorId, name, phone } = req.body;
  const job = jobsData.find(j => j.id === req.params.id);
  if (!job) return res.status(404).json({ success: false, message: '공고를 찾을 수 없습니다.' });
  if (!job.candidates) job.candidates = [];
  if (job.candidates.find(c => c.seniorId === seniorId)) return res.status(400).json({ success: false, message: '이미 지원한 공고입니다.' });

  job.candidates.push({ seniorId, name, phone, status: '지원 완료 (사장님 검토 중)', rank: null, bonusPayPerHour: 0 });
  res.json({ success: true });
});

app.post('/api/jobs/:id/cancel-hire', (req, res) => {
  const { seniorId } = req.body;
  const job = jobsData.find(j => j.id === req.params.id);
  if (!job) return res.status(404).json({ success: false, message: '공고를 찾을 수 없습니다.' });

  const cancelledCand = job.candidates.find(c => c.seniorId === seniorId);
  if (cancelledCand) {
    cancelledCand.status = '채용 취소됨';
    notificationsData.unshift({
      id: 'noti-' + Date.now(), targetUserId: seniorId, title: '[채용 취소 알림]',
      content: `'${job.title}' 공고의 채용이 취소되었습니다.`, date: new Date().toLocaleString(), isRead: false
    });
  }

  const candidate2 = job.candidates.find(c => c.rank === '2순위' && c.status !== '채용 취소됨');
  const candidate3 = job.candidates.find(c => c.rank === '3순위' && c.status !== '채용 취소됨');
  let promotedCand = candidate2 || candidate3;

  if (promotedCand) {
    const isSecond = promotedCand.rank === '2순위';
    const bonusPerHour = isSecond ? 1000 : 2000;
    promotedCand.status = `${promotedCand.rank} 승계 채용 확정 (시급 +${bonusPerHour.toLocaleString()}원 가산)`;
    promotedCand.bonusPayPerHour = bonusPerHour;

    notificationsData.unshift({
      id: 'noti-' + Date.now(), targetUserId: promotedCand.seniorId, title: '[자동 승계 채용 알림]',
      content: `'${job.title}' 공고에 1순위 취소로 인해 ${promotedCand.rank} 승계 채용되었습니다! (시급 당 +${bonusPerHour.toLocaleString()}원 추가 가산)`,
      date: new Date().toLocaleString(), isRead: false
    });
    notificationsData.unshift({
      id: 'noti-' + Date.now(), targetUserId: job.employerId, title: '[승계 채용 처리 알림]',
      content: `'${job.title}' 공고의 1순위 채용 취소로 인해 ${promotedCand.name}님(${promotedCand.rank})이 자동 승계 채용되었습니다.`,
      date: new Date().toLocaleString(), isRead: false
    });
  }
  res.json({ success: true });
});

// 포인트 지급 시 임금의 10% (구독 시 5%) 명세서 금액 자동 발행
app.post('/api/jobs/:id/pay-points', (req, res) => {
  const { seniorId, basePay } = req.body;
  const job = jobsData.find(j => j.id === req.params.id);
  const senior = usersData.find(u => u.id === seniorId);
  const employer = usersData.find(u => u.id === job?.employerId);

  if (job && senior) {
    const cand = job.candidates.find(c => c.seniorId === seniorId);
    let bonusTotal = 0;
    if (cand && cand.bonusPayPerHour > 0) bonusTotal = cand.bonusPayPerHour * 3;

    const totalPay = parseInt(basePay, 10) + bonusTotal;
    senior.points = (senior.points || 0) + totalPay;

    const feeRate = employer && employer.isSubscribed ? 0.05 : 0.10;
    const feeAmount = Math.round(totalPay * (1 + feeRate)); // 일당의 10%(또는 5%)가 붙은 총액 명세서

    invoicesData.unshift({
      id: 'inv-' + Date.now(),
      employerId: job.employerId,
      jobTitle: job.title,
      storeName: job.storeName,
      amount: feeAmount,
      date: new Date().toLocaleString(),
      status: 'ISSUED', // ISSUED, PENDING_ADMIN, CONFIRMED
      isFullyConfirmed: false
    });

    notificationsData.unshift({
      id: 'noti-' + Date.now(), targetUserId: seniorId, title: '[포인트 지급 완료]',
      content: `'${job.title}' 근무 대가로 ${totalPay.toLocaleString()}P가 지급되었습니다.`, date: new Date().toLocaleString(), isRead: false
    });

    notificationsData.unshift({
      id: 'noti-' + Date.now(), targetUserId: job.employerId, title: '[수수료 명세서 발행]',
      content: `'${job.title}' 채용에 따른 수수료 납부 명세서(${feeAmount.toLocaleString()}원)가 발행되었습니다.`, date: new Date().toLocaleString(), isRead: false
    });
  }
  res.json({ success: true });
});

app.post('/api/settlements/:id/complete', (req, res) => {
  const settle = settlementsData.find(s => s.id === req.params.id);
  if (settle && settle.status === 'PENDING') {
    settle.status = 'COMPLETED';
    const user = usersData.find(u => u.id === settle.userId);
    if (user) {
      user.points = Math.max(0, user.points - settle.amount);
      notificationsData.unshift({
        id: 'noti-' + Date.now(), targetUserId: user.id, title: '[정산 완료 알림]',
        content: `요청하신 정산 금액 ${settle.amount.toLocaleString()}P 처리가 완료되었습니다.`, date: new Date().toLocaleString(), isRead: false
      });
    }
  }
  res.json({ success: true });
});

        function openCancelModal(jobId, seniorId) {
            document.getElementById('cancel-job-id').value = jobId;
            document.getElementById('cancel-senior-id').value = seniorId;
            document.getElementById('cancel-agree-check').checked = false;
            document.getElementById('cancelHireModal').classList.remove('hidden');
        }
        function closeCancelModal() { document.getElementById('cancelHireModal').classList.add('hidden'); }

        async function confirmCancelHire() {
            const isAgreed = document.getElementById('cancel-agree-check').checked;
            if (!isAgreed) {
                alert('본사 접수 동의 체크박스에 동의해 주셔야 채용 취소 및 자동 승계 처리가 진행됩니다.');
                return;
            }

            const jobId = document.getElementById('cancel-job-id').value;
            const seniorId = document.getElementById('cancel-senior-id').value;

            await fetch(`/api/jobs/${jobId}/cancel-hire`, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ seniorId })
            }).then(r => r.json());

            closeCancelModal();

            alert('회사 신고(노쇼 접수)가 정상 처리되었습니다.\n다음 순위 지원자에게 자동 승계 채용 알림이 발송되었습니다.');
            renderProfile();
        }

        function openPayModal(jobId, seniorId, amount) {
            document.getElementById('pay-job-id').value = jobId;
            document.getElementById('pay-senior-id').value = seniorId;
            document.getElementById('pay-amount').value = amount;
            document.getElementById('pay-agree-check').checked = false;
            document.getElementById('payPointModal').classList.remove('hidden');
        }
        function closePayModal() { document.getElementById('payPointModal').classList.add('hidden'); }

        async function confirmPayPoint() {
            const isAgreed = document.getElementById('pay-agree-check').checked;
            if (!isAgreed) {
                alert('시니어 회원의 실제 근무 확인 동의란을 체크해 주셔야 안전한 포인트 지급이 가능합니다.');
                return;
            }

            const jobId = document.getElementById('pay-job-id').value;
            const seniorId = document.getElementById('pay-senior-id').value;
            const basePay = document.getElementById('pay-amount').value;

            await fetch(`/api/jobs/${jobId}/pay-points`, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ seniorId, basePay })
            });

            closePayModal();
            alert('포인트 지급이 완료되었으며 수수료 납부 명세서가 발행되었습니다.');
            renderProfile();
        }

        async function assignRank(jobId, seniorId, rank) {
            if (!rank) return;
            await fetch(`/api/jobs/${jobId}/rank`, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ seniorId, rank })
            });
            alert(`${rank} 지정 완료 및 시니어에게 알림이 발송되었습니다.`);
            renderProfile();
        }

        async function hireCandidate(jobId, seniorId) {
            await fetch(`/api/jobs/${jobId}/hire`, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ seniorId })
            });
            alert('채용 확정 및 시니어 회원에게 알림이 발송되었습니다.');
            renderProfile();
        }

        function updateCardClasses() {
            const cards = document.querySelectorAll('.shuffle-card');
            cards.forEach((card, index) => {
                const posIndex = cardOrder.indexOf(index);
                card.className = `shuffle-card card-pos-${posIndex} space-y-4`;
            });

        function setFrontCard(targetIndex) {
            const currentPos = cardOrder.indexOf(targetIndex);
            if (currentPos === 1) {
                cardOrder = [cardOrder[1], cardOrder[2], cardOrder[0]];
            } else if (currentPos === 2) {
                cardOrder = [cardOrder[2], cardOrder[0], cardOrder[1]];
            }
            updateCardClasses();
        }

        async function handleInquirySubmit(e) {
            e.preventDefault();
            const name = document.getElementById('inq-name').value;
            const storeName = document.getElementById('inq-store').value;
            const phone = document.getElementById('inq-phone').value;

            try {
                const res = await fetch('/api/inquiries', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ name, storeName, phone })
                }).then(r => r.json());

                if (res.success) {
                    alert('1시간 내로 실버웍스 측에서 전화가 갈 예정입니다. 신청해 주셔서 감사합니다.');

                    document.getElementById('inq-store').value = '';
                    document.getElementById('inq-phone').value = '';
                } else {
                    alert('문의 접수 중 오류가 발생했습니다.');
                }
            } catch (err) {
                console.error(err);

        async function updateHeaderUI() {
            const userArea = document.getElementById('header-user-area');
            const user = JSON.parse(localStorage.getItem('sw_user') || 'null');

            if (!user) {
                userArea.innerHTML = `<a href="/login.html" class="text-xs font-bold px-4 py-2 rounded-xl bg-sky-500 text-white hover:bg-sky-600 transition shadow-sm">로그인 및 회원가입</a>`;
                return;
            }

            const usersRes = await fetch('/api/users').then(r => r.json());
            const freshUser = usersRes.users.find(u => u.id === user.id) || user;
            localStorage.setItem('sw_user', JSON.stringify(freshUser));

            const notiRes = await fetch(`/api/notifications/${user.id}`).then(r => r.json());
            const hasUnread = notiRes.hasUnread;

            let html = `
                <a href="/notifications.html" class="relative p-2 text-slate-500 hover:text-sky-600 transition" title="알림 센터">
                    <svg class="w-6 h-6" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round">
                        <path d="M18 8A6 6 0 0 0 6 8c0 7-3 9-3 9h18s-3-2-3-9"></path>
                        <path d="M13.73 21a2 2 0 0 1-3.46 0"></path>
                    </svg>
                    ${hasUnread ? `<span class="absolute top-1 right-1 w-2.5 h-2.5 bg-sky-500 rounded-full ring-2 ring-white"></span>` : ''}
                </a>
            `;

            if (freshUser.role === 'senior') {
                html += `
                    <a href="/profile.html" class="text-xs font-extrabold bg-sky-50 text-sky-600 px-3 py-1.5 rounded-xl border border-sky-100 hover:bg-sky-100 transition">
                        포인트 ${(freshUser.points || 0).toLocaleString()}P (정산)
                    </a>
                `;
            }

            html += `
                <a href="/profile.html" class="w-9 h-9 rounded-full bg-sky-100 text-sky-700 flex items-center justify-center font-bold border border-sky-200 hover:border-sky-400 transition" title="프로필 페이지">

                </a>
                <button onclick="handleLogout()" class="text-xs font-bold text-slate-500 hover:text-rose-500 px-2 py-1.5 rounded-lg transition">로그아웃</button>
            `;
            userArea.innerHTML = html;
        }

        async function loadAdminControl() {
            const usersRes = await fetch('/api/users').then(r => r.json());
            const setRes = await fetch('/api/settlements').then(r => r.json());
            const inqRes = await fetch('/api/inquiries').then(r => r.json());
            const invRes = await fetch('/api/invoices').then(r => r.json());

            const users = usersRes.users || [];
            const settlements = setRes.settlements || [];
            const inquiries = inqRes.inquiries || [];
            const invoices = invRes.invoices || [];

            // 1. Invoices Admin Control List
            const invContainer = document.getElementById('admin-invoice-list');
            const pendingInvoices = invoices.filter(i => i.status === 'PENDING_ADMIN' && !i.isFullyConfirmed);
            if (pendingInvoices.length === 0) {
                invContainer.innerHTML = `<p class="text-xs text-slate-400 text-center py-6">사장님이 확인 요청한 수수료 명세서가 없습니다.</p>`;
            } else {
                invContainer.innerHTML = pendingInvoices.map(i => `
                    <div class="p-4 rounded-2xl bg-sky-50/50 border border-sky-100 flex justify-between items-center text-xs">
                        <div>
                            <strong class="text-sm font-bold text-slate-800">${i.storeName} (${i.jobTitle})</strong>
                            <p class="text-slate-500 mt-0.5">청구 수수료: ${i.amount.toLocaleString()}원 | 발행일시: ${i.date}</p>
                        </div>
                        <button onclick="adminConfirmInvoice('${i.id}')" class="btn-sky font-bold px-4 py-2 rounded-xl shadow-sm">
                            확인 (사장님 명세서 소멸)
                        </button>
                    </div>
                `).join('');
            }

            // 2. Inquiries Table
            const inqTable = document.getElementById('inquiry-table-body');
            if (inquiries.length === 0) {
                inqTable.innerHTML = `<tr><td colspan="4" class="p-6 text-center text-slate-400">신청된 전화 상담 문의가 없습니다.</td></tr>`;
            } else {
                inqTable.innerHTML = inquiries.map(i => `
                    <tr>
                        <td class="p-3 text-slate-400">${i.date}</td>
                        <td class="p-3 font-bold text-slate-800">${i.name}</td>
                        <td class="p-3 font-bold text-sky-600">${i.storeName}</td>
                        <td class="p-3 font-mono font-bold">${i.phone}</td>
                    </tr>
                `).join('');
            }

            // 3. User Approval Table
            const userTable = document.getElementById('user-approval-table');
            if (users.length === 0) {
                userTable.innerHTML = `<tr><td colspan="7" class="p-6 text-center text-slate-400">가입된 회원이 없습니다.</td></tr>`;
            } else {
                userTable.innerHTML = users.map(u => `
                    <tr>
                        <td class="p-3 font-mono font-bold text-sky-600 cursor-pointer hover:underline" onclick="viewUserDetail('${u.id}')">${u.id}</td>
                        <td class="p-3 font-bold">${u.name}</td>
                        <td class="p-3"><span class="px-2 py-0.5 rounded text-[11px] font-bold ${u.role === 'senior' ? 'bg-sky-100 text-sky-700' : 'bg-indigo-100 text-indigo-700'}">${u.role === 'senior' ? '시니어' : '사장님'}</span></td>
                        <td class="p-3">
                            ${u.role === 'senior' ? `<button onclick="editSeniorPoints('${u.id}', ${u.points || 0})" class="bg-sky-50 text-sky-600 font-extrabold px-2 py-1 rounded border border-sky-100 hover:bg-sky-100 text-[11px]">${(u.points || 0).toLocaleString()}P <i class="fa-solid fa-pen text-[9px] ml-0.5"></i></button>` : '<span class="text-slate-300">-</span>'}
                        </td>
                        <td class="p-3"><span class="font-bold ${u.approved ? 'text-emerald-500' : 'text-amber-500'}">${u.approved ? '승인완료' : '승인대기'}</span></td>
                        <td class="p-3"><span class="font-bold ${u.isSubscribed ? 'text-sky-600' : 'text-slate-400'}">${u.isSubscribed ? '구독회원' : (u.subRequested ? '구독신청중' : '일반')}</span></td>
                        <td class="p-3 text-right space-x-1">
                            ${!u.approved ? `<button onclick="approveUser('${u.id}')" class="btn-sky text-[11px] px-2.5 py-1 rounded-lg font-bold">승인</button>` : ''}
                            ${u.role === 'employer' && u.subRequested && !u.isSubscribed ? `<button onclick="approveSub('${u.id}')" class="bg-emerald-500 text-white text-[11px] px-2.5 py-1 rounded-lg font-bold hover:bg-emerald-600">구독승인</button>` : ''}
                            ${u.role === 'employer' ? `
                                <button onclick="sendWarning('${u.id}', 1)" class="bg-amber-100 text-amber-700 text-[10px] px-1.5 py-1 rounded font-bold">1차경고</button>
                                <button onclick="sendWarning('${u.id}', 2)" class="bg-amber-500 text-white text-[10px] px-1.5 py-1 rounded font-bold">2차경고</button>
                                <button onclick="sendWarning('${u.id}', 3)" class="bg-rose-500 text-white text-[10px] px-1.5 py-1 rounded font-bold">3차경고</button>
                            ` : ''}
                            <button onclick="forceDeleteUser('${u.id}')" class="bg-rose-100 text-rose-600 text-[11px] px-2 py-1 rounded-lg font-bold hover:bg-rose-200">강제탈퇴</button>
                        </td>
                    </tr>
                `).join('');
            }

            // 4. Settlements Table
            const setList = document.getElementById('admin-settlement-list');
            const pendingSettles = settlements.filter(s => s.status === 'PENDING');
            if (pendingSettles.length === 0) {
                setList.innerHTML = `<p class="text-xs text-slate-400 text-center py-6">대기 중인 정산 요청이 없습니다.</p>`;
            } else {
                setList.innerHTML = pendingSettles.map(s => `
                    <div class="p-4 rounded-2xl bg-amber-50/50 border border-amber-100 flex justify-between items-center text-xs">
                        <div>
                            <strong class="text-sm font-bold text-slate-800">${s.userName} (${s.userId})</strong>
                            <p class="text-slate-500 mt-0.5">입금계좌: ${s.bank} ${s.account} | 요청금액: ${s.amount.toLocaleString()}P</p>
                        </div>
                        <button onclick="completeSettlement('${s.id}')" class="bg-amber-500 text-white font-bold px-4 py-2 rounded-xl shadow-sm hover:bg-amber-600">
                            정산 완료
                        </button>
                    </div>
                `).join('');
            }
        }

        async function adminConfirmInvoice(invoiceId) {
            if (confirm('해당 명세서 처리를 완료하시겠습니까? 사장님 페이지에서 명세서가 사라집니다.')) {
                await fetch('/api/invoices/admin-confirm', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ invoiceId })
                });
                alert('명세서 처리가 완료되었습니다.');
                loadAdminControl();
            }
        }

        async function editSeniorPoints(userId, currentP) {
            const newP = prompt('수정할 보유 포인트 액수를 입력하세요:', currentP);
            if (newP !== null && !isNaN(newP)) {
                await fetch(`/api/users/${userId}/update-points`, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ points: parseInt(newP, 10) })
                });
                alert('시니어 회원의 보유 포인트가 변경되었습니다.');
                loadAdminControl();
            }
        }

        async function viewUserDetail(id) {
            const usersRes = await fetch('/api/users').then(r => r.json());
            const u = usersRes.users.find(user => user.id === id);
            const modalBody = document.getElementById('user-modal-body');
            
            modalBody.innerHTML = `
                <p><strong>아이디:</strong> ${u.id}</p>
                <p><strong>성함/대표자명:</strong> ${u.name}</p>
                <p><strong>연락처:</strong> ${u.phone || '미입력'}</p>
                <p><strong>회원 유형:</strong> ${u.role === 'senior' ? '시니어 구직자' : '구인 사장님'}</p>
                <p><strong>가입 승인 상태:</strong> ${u.approved ? '승인 완료' : '승인 대기중'}</p>
                <p><strong>구독 회원 여부:</strong> ${u.isSubscribed ? '구독 회원' : '일반 회원'}</p>
                <p><strong>보유 포인트:</strong> ${(u.points || 0).toLocaleString()}P</p>
            `;
            document.getElementById('userDetailModal').classList.remove('hidden');
        }
        function closeUserModal() { document.getElementById('userDetailModal').classList.add('hidden'); }

        async function approveUser(id) {
            await fetch(`/api/users/${id}/approve`, { method: 'POST' });
            alert('가입 승인이 완료되었습니다.');
            loadAdminControl();
        }

        async function approveSub(id) {
            await fetch(`/api/users/${id}/approve-subscription`, { method: 'POST' });
            alert('사장님 구독이 승인되었습니다.');
            loadAdminControl();
        }

        async function cancelSub(id) {
            if (confirm('해당 사장님의 구독권을 해지하시겠습니까?')) {
                await fetch(`/api/users/${id}/cancel-subscription`, { method: 'POST' });
                alert('구독권이 해지되었습니다.');
                loadAdminControl();
            }
        }

        async function sendWarning(userId, step) {
            if (confirm(`해당 사장님에게 ${step}차 경고 알림을 발송하시겠습니까?`)) {
                await fetch(`/api/users/${userId}/warn`, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ step })
                });
                alert(`${step}차 경고 알림이 발송되었습니다.`);
            }
        }

        async function forceDeleteUser(userId) {
            if (confirm(`[주의] ${userId} 회원을 정말 강제 탈퇴(계정 삭제) 처리하시겠습니까?`)) {
                await fetch(`/api/users/${userId}`, { method: 'DELETE' });
                alert('회원이 강제 탈퇴 처리되었습니다.');
                loadAdminControl();
            }
        }

        async function completeSettlement(settleId) {
            if (confirm('정산을 완료하시겠습니까? 시니어 회원에게 알림이 전송됩니다.')) {
                await fetch(`/api/settlements/${settleId}/complete`, { method: 'POST' });
                alert('정산 완료 처리되었습니다.');
                loadAdminControl();
            }
        }

        document.addEventListener('DOMContentLoaded', loadAdminControl);
    </script>
