// 공고 목록 조회
app.get('/api/jobs', (req, res) => {
    const { category, search } = req.query;
    let filtered = [...jobsData];

    if (category && category !== '전체') {
        filtered = filtered.filter(j => j.category === category);
    }
    if (search) {
        filtered = filtered.filter(j => j.title.includes(search) || j.storeName.includes(search));
    }

    res.json({ success: true, count: filtered.length, jobs: filtered });
});

// 단일 공고 상세 조회
app.get('/api/jobs/:id', (req, res) => {
    const job = jobsData.find(j => j.id === req.params.id);
    if (!job) return res.status(404).json({ success: false, message: '공고를 찾을 수 없습니다.' });
    res.json({ success: true, job });
});

// 신규 공고 등록
app.post('/api/jobs', (req, res) => {
    const { storeName, title, category, badgeReq, date, time, pay, isSubscribed } = req.body;
    const newJob = {
        id: 'job-' + Date.now(),
        storeName: storeName || '대박식당',
        title,
        category: category || '식당·서빙',
        badgeReq: badgeReq || '체력검증 우대',
        date,
        time,
        pay: parseInt(pay, 10),
        feeRate: isSubscribed ? 0.05 : 0.10,
        status: '모집중',
        candidates: []
    };
    jobsData.unshift(newJob);
    res.status(201).json({ success: true, job: newJob });
});

// 지원하기 (1·2·3순위 대기열)
app.post('/api/jobs/:id/apply', (req, res) => {
    const { applicantName, phone, badge } = req.body;
    const job = jobsData.find(j => j.id === req.params.id);
    if (!job) return res.status(404).json({ success: false, message: '공고가 없습니다.' });

    const rank = job.candidates.length + 1;
    if (rank > 3) return res.status(400).json({ success: false, message: '대기열이 가득 찼습니다.' });

    let statusText = `${rank}순위 출근확정`;
    if (rank === 2) statusText = '2순위 대기중 (+5,000원 가산 대상)';
    if (rank === 3) statusText = '3순위 예비대기';

    const newApplicant = { id: 'u-' + Date.now(), name: applicantName || '김시니어', phone, badge, rank, status: statusText };
    job.candidates.push(newApplicant);

    res.json({ success: true, rank, applicant: newApplicant });
});

// 노쇼 처리 및 2순위 자동 승계
app.post('/api/jobs/:id/noshow', (req, res) => {
    const job = jobsData.find(j => j.id === req.params.id);
    if (!job || job.candidates.length < 2) {
        return res.status(400).json({ success: false, message: '승계할 2순위 대기자가 없습니다.' });
    }

    job.candidates.shift(); // 1순위 제거
    job.candidates.forEach((c, idx) => {
        c.rank = idx + 1;
        c.status = c.rank === 1 ? '2순위 승계 출근확정 (+5,000원 반영)' : `${c.rank}순위 대기`;
    });
    job.status = '2순위 승계 완료';

    res.json({ success: true, promotedCandidate: job.candidates[0] });
});

        function triggerNoShow(jobId) {
            if (!confirm('1순위 지원자의 노쇼를 확정하시겠습니까?\n즉시 2순위 대기자에게 승계 및 가산금(+5,000원)이 부여됩니다.')) return;

            const jobs = JSON.parse(localStorage.getItem('sw_jobs')) || [];
            const job = jobs.find(j => j.id === jobId);

            if (job && job.candidates && job.candidates.length > 1) {
                // Remove Rank 1 (No-show), promote Rank 2 -> Rank 1
                job.candidates.shift(); // Remove rank 1
                job.candidates.forEach((c, idx) => {
                    c.rank = idx + 1;
                    if (c.rank === 1) {
                        c.status = '2순위 승계 출근확정 (가산금 적용)';
                    } else {
                        c.status = `${c.rank}순위 대기`;
                    }
                });
                job.status = '2순위 자동 승계 완료';

                localStorage.setItem('sw_jobs', JSON.stringify(jobs));
                alert('[노쇼 승계 성공]\n\n2순위 대기자에게 긴급 알림이 발송되었으며 승계 출근이 확정되었습니다.');
                loadEmployerJobs();
            } else {
                alert('승계할 2순위 대기 지원자가 없습니다.');
            }
        }

        let currentJob = null;

        function getJobIdFromUrl() {
            const params = new URLSearchParams(window.location.search);
            return params.get('id') || 'job-1';
        }

        function loadJobDetail() {
            const jobId = getJobIdFromUrl();
            const jobs = JSON.parse(localStorage.getItem('sw_jobs')) || [];
            currentJob = jobs.find(j => j.id === jobId);

            if (!currentJob) {
                alert('존재하지 않는 공고입니다.');
                window.location.href = '/jobs.html';
                return;
            }

            // Populate UI Elements
            document.getElementById('detail-category').innerText = currentJob.category;
            document.getElementById('detail-badge-req').innerText = currentJob.badgeReq;
            document.getElementById('detail-status').innerText = currentJob.status;
            document.getElementById('detail-store').innerText = currentJob.storeName;
            document.getElementById('detail-title').innerText = currentJob.title;
            document.getElementById('detail-date').innerText = currentJob.date;
            document.getElementById('detail-time').innerText = currentJob.time;
            document.getElementById('detail-pay').innerText = currentJob.pay.toLocaleString() + '원';

            renderCandidates(currentJob.candidates || []);
        }

        function renderCandidates(candidates) {
            const container = document.getElementById('candidate-list-container');
            const ranks = [1, 2, 3];

            container.innerHTML = ranks.map(rank => {
                const candidate = candidates.find(c => c.rank === rank);

                if (candidate) {
                    return `
                        <div class="flex items-center justify-between p-4 rounded-2xl bg-blue-50/50 border border-blue-100/80">
                            <div class="flex items-center gap-3">
                                <span class="w-8 h-8 rounded-xl bg-blue-600 text-white font-extrabold text-xs flex items-center justify-center">${rank}순위</span>
                                <div>
                                    <strong class="text-xs text-slate-900 font-extrabold block">${candidate.name} (${candidate.age}세)</strong>
                                    <span class="text-[11px] text-slate-500">배지: ${candidate.badge} | 신청시간 ${candidate.time}</span>
                                </div>
                            </div>
                            <span class="text-xs font-bold text-blue-600 bg-white px-3 py-1 rounded-lg border border-blue-200">
                                ${candidate.status}
                            </span>
                        </div>
                    `;
                } else {
                    return `
                        <div class="flex items-center justify-between p-4 rounded-2xl bg-slate-50 border border-dashed border-slate-200">
                            <div class="flex items-center gap-3">
                                <span class="w-8 h-8 rounded-xl bg-slate-200 text-slate-600 font-bold text-xs flex items-center justify-center">${rank}순위</span>
                                <span class="text-xs text-slate-400 font-medium">대기열 비어있음 (지원 가능)</span>
                            </div>
                            <span class="text-xs font-bold text-slate-400 bg-white px-3 py-1 rounded-lg border border-slate-200">
                                모집중
                            </span>
                        </div>
                    `;
                }
            }).join('');
        }

        function openApplyModal() {
            document.getElementById('applyModal').classList.remove('hidden');
        }

        function closeApplyModal() {
            document.getElementById('applyModal').classList.add('hidden');
        }

        function submitApplication(e) {
            e.preventDefault();
            const name = document.getElementById('app-name').value;
            const badge = document.getElementById('app-badge').value;

            if (!currentJob.candidates) {
                currentJob.candidates = [];
            }

            if (currentJob.candidates.length >= 3) {
                alert('해당 공고의 1·2·3순위 대기열이 모두 마감되었습니다.');
                closeApplyModal();
                return;
            }

            const nextRank = currentJob.candidates.length + 1;
            const now = new Date();
            const timeStr = `${String(now.getHours()).padStart(2, '0')}:${String(now.getMinutes()).padStart(2, '0')}`;

            const newCandidate = {
                rank: nextRank,
                name: name,
                age: 66,
                time: timeStr,
                badge: badge,
                status: nextRank === 1 ? '출근대기' : `${nextRank}순위 대기`
            };

            currentJob.candidates.push(newCandidate);
            if (nextRank === 1) {
                currentJob.status = '1순위 매칭 완료';
            }

            // Save back to LocalStorage
            const jobs = JSON.parse(localStorage.getItem('sw_jobs')) || [];
            const idx = jobs.findIndex(j => j.id === currentJob.id);
            if (idx !== -1) {
                jobs[idx] = currentJob;
                localStorage.setItem('sw_jobs', JSON.stringify(jobs));
            }

            alert(`[지원 완료]\n\n${name}님, ${nextRank}순위 지원이 성공적으로 완료되었습니다.`);
            closeApplyModal();
            loadJobDetail();
        }

        document.addEventListener('DOMContentLoaded', loadJobDetail);
