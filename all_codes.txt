function addLiveLog(type, message) {
  const timestamp = new Date().toLocaleString('ko-KR');
  liveLogsData.unshift({ id: 'log-' + Date.now(), type, message, timestamp });
  if (liveLogsData.length > 50) liveLogsData.pop();
}

app.post('/api/signup', (req, res) => {
  const { id, pw, name, phone, role } = req.body;
  if (!id || !pw || !name || !role) return res.status(400).json({ success: false, message: '필수 회원가입 정보가 누락되었습니다.' });
  if (pw.length < 8) return res.status(400).json({ success: false, message: '비밀번호는 최소 8자리 이상이어야 합니다.' });
  if (usersData.find(u => u.id === id)) return res.status(400).json({ success: false, message: '이미 존재하는 아이디입니다.' });

  const newUser = { id, pw, name, phone: phone || '', role, approved: false, points: 0, isSubscribed: false, subRequested: false, hasSurvey: false, isVeteran: false };
  usersData.push(newUser);

  notificationsData.unshift({
    id: 'noti-' + Date.now(),
    targetUserId: id,
    title: '[회원가입 축하]',
    content: `${name}님, silverworks 회원가입을 진심으로 축하합니다!`,
    date: new Date().toLocaleString('ko-KR'),
    isRead: false
  });

  addLiveLog('SIGNUP', `[신규가입] ${name}(${role === 'senior' ? '시니어' : '사장님'}) 님이 가입 신청했습니다.`);
  res.json({ success: true, user: newUser });
});

app.post('/api/login', (req, res) => {
  const { id, pw, role } = req.body;
  if ((id === 'silverworks' && pw === 'silverworks1@') || (id === 'admin' && pw === '1234')) {
    const adminUser = usersData.find(u => u.id === id) || { id, pw, name: '최고관리자', role: 'admin', approved: true, points: 0, isSubscribed: true, hasSurvey: true, isVeteran: true };
    return res.json({ success: true, user: adminUser });
  }

  const user = usersData.find(u => u.id === id && u.role === role);
  if (!user) return res.status(400).json({ success: false, message: '등록되지 않았거나 회원 유형이 일치하지 않습니다.' });
  if (user.pw !== pw) return res.status(400).json({ success: false, message: '비밀번호가 일치하지 않습니다.' });
  if (!user.approved) return res.status(400).json({ success: false, message: '관리자의 가입 승인 대기 중입니다.' });

  addLiveLog('LOGIN', `[로그인] ${user.name}(${user.id}) 님이 접속했습니다.`);
  res.json({ success: true, user });
});

app.post('/api/jobs', (req, res) => {
  const { employerId, storeName, title, category, date, time, pay, isPeak, regionMain, regionSub, address, description } = req.body;
  const employer = usersData.find(u => u.id === employerId);
  if (isPeak && (!employer || !employer.isSubscribed)) {
    return res.status(403).json({ success: false, message: '상단 고정 공고는 구독 회원 사장님만 이용 가능합니다.' });
  }

  const newJob = {
    id: 'job-' + Date.now(),
    employerId,
    storeName,
    title: isPeak ? `[상단 고정] ${title}` : title,
    category: category || '음식',
    date,
    time,
    pay: parseInt(pay, 10),
    isPeak: !!isPeak,
    regionMain: regionMain || '서울특별시',
    regionSub: regionSub || '전체',
    address: address || '',
    description: description || '',
    status: '구직자 모집중',
    candidates: []
  };

  jobsData.unshift(newJob);
  addLiveLog('JOB_CREATE', `[공고등록] ${storeName}에서 '${newJob.title}' 구인 공고를 등록했습니다.`);
  res.json({ success: true, jobId: newJob.id });
});

app.delete('/api/jobs/:id', (req, res) => {
  const jobIdx = jobsData.findIndex(j => j.id === req.params.id);
  if (jobIdx !== -1) {
    const job = jobsData[jobIdx];
    if (job.candidates && job.candidates.length > 0) {
      job.candidates.forEach(c => {
        notificationsData.unshift({
          id: 'noti-' + Date.now() + '-' + Math.random(),
          targetUserId: c.seniorId,
          title: '[공고 삭제 알림]',
          content: `지원하신 '${job.title}' 공고가 사장님 또는 관리자에 의해 삭제되었습니다.`,
          date: new Date().toLocaleString('ko-KR'),
          isRead: false
        });
      });
    }
    jobsData.splice(jobIdx, 1);
    addLiveLog('JOB_DELETE', `[공고삭제] 공고 ID(${req.params.id})가 삭제 처리되었습니다.`);
  }
  res.json({ success: true });
});

app.post('/api/jobs/:id/apply', (req, res) => {
  const { seniorId, name, phone } = req.body;
  const job = jobsData.find(j => j.id === req.params.id);
  if (!job) return res.status(404).json({ success: false, message: '공고를 찾을 수 없습니다.' });

  if (!job.candidates) job.candidates = [];
  if (job.candidates.find(c => c.seniorId === seniorId)) return res.status(400).json({ success: false, message: '이미 지원한 공고입니다.' });

  job.candidates.push({ seniorId, name, phone, status: '지원 완료 (사장님 검토 중)', rank: null });
  addLiveLog('APPLY', `[공고지원] 시니어 ${name}(${seniorId})님이 공고에 지원했습니다.`);
  res.json({ success: true });
});

// 명세서 금액 계산 공식 전면 수정: 기본일당 + 수수료 + 부가가치세 = 최종 청구액
app.post('/api/jobs/:id/pay-points', (req, res) => {
  const { seniorId, basePay } = req.body;
  const job = jobsData.find(j => j.id === req.params.id);
  const senior = usersData.find(u => u.id === seniorId);
  const employer = usersData.find(u => u.id === job?.employerId);

  if (job && senior) {
    const payAmount = parseInt(basePay, 10);
    senior.points = (senior.points || 0) + payAmount;

    const feeRate = employer && employer.isSubscribed ? 0.05 : 0.10;
    const feeAmount = Math.round(payAmount * feeRate);
    const vat = Math.round(feeAmount * 0.10);
    // 수정: 기본일당 + 수수료 + VAT = 최종 청구액
    const totalAmount = payAmount + feeAmount + vat;

    invoicesData.unshift({
      id: 'inv-' + Date.now(),
      employerId: job.employerId,
      jobId: job.id,
      jobTitle: job.title,
      storeName: job.storeName,
      seniorId,
      workDate: job.date,
      basePay: payAmount,
      feeRate,
      feeAmount,
      vat,
      totalAmount,
      date: new Date().toLocaleString('ko-KR'),
      status: 'ISSUED',
      isFullyConfirmed: false
    });

    notificationsData.unshift({
      id: 'noti-' + Date.now() + '-1',
      targetUserId: seniorId,
      title: '[포인트 지급 완료]',
      content: `'${job.title}' 근무 대가로 ${payAmount.toLocaleString()}P가 지급되었습니다.`,
      date: new Date().toLocaleString('ko-KR'),
      isRead: false
    });

    notificationsData.unshift({
      id: 'noti-' + Date.now() + '-2',
      targetUserId: job.employerId,
      title: '[수수료 명세서 발행]',
      content: `'${job.title}' 채용에 따른 수수료 명세서(최종 청구액 ${totalAmount.toLocaleString()}원)가 발행되었습니다. (사장님이 시니어에게 직접 현금을 지급하지 않아도 플랫폼을 통해 안전 정산됩니다)`,
      date: new Date().toLocaleString('ko-KR'),
      isRead: false
    });

    addLiveLog('PAY_POINTS', `[포인트지급] ${job.storeName}에서 시니어(${seniorId})에게 ${payAmount.toLocaleString()}P 지급 및 명세서 발행 완료.`);
  }
  res.json({ success: true });
});

  if (!user || user.points < amount) return res.status(400).json({ success: false, message: '보유 포인트가 부족합니다.' });

  const newSettle = { id: 'settle-' + Date.now(), userId, userName, bank, account, amount: parseInt(amount, 10), status: 'PENDING', date: new Date().toLocaleString('ko-KR') };
  settlementsData.unshift(newSettle);
  addLiveLog('SETTLE_REQ', `[정산요청] 시니어 ${userName}(${userId})님이 ${parseInt(amount).toLocaleString()}P 출금 정산을 요청했습니다.`);
  res.json({ success: true, settlement: newSettle });
});

app.post('/api/settlements/:id/complete', (req, res) => {
  const settle = settlementsData.find(s => s.id === req.params.id);
  if (settle && settle.status === 'PENDING') {
    settle.status = 'COMPLETED';
    const user = usersData.find(u => u.id === settle.userId);
    if (user) {
      user.points = Math.max(0, user.points - settle.amount);
      notificationsData.unshift({
        id: 'noti-' + Date.now(),
        targetUserId: user.id,
        title: '[정산 완료 알림]',
        content: `요청하신 정산 금액 ${settle.amount.toLocaleString()}원(${settle.bank}) 입금 처리가 완료되었습니다.`,
        date: new Date().toLocaleString('ko-KR'),
        isRead: false
      });
      addLiveLog('SETTLE_OK', `[정산완료] 관리자가 시니어(${settle.userId})의 ${settle.amount.toLocaleString()}P 정산을 완료했습니다.`);
    }
  }
  res.json({ success: true });
});

app.get('/api/notifications/:userId', (req, res) => {
  const userNotis = notificationsData.filter(n => n.targetUserId === req.params.userId);
  const hasUnread = userNotis.some(n => !n.isRead);
  res.json({ success: true, notifications: userNotis, hasUnread });
});

app.post('/api/notifications/:userId/read', (req, res) => {
  notificationsData.forEach(n => { if (n.targetUserId === req.params.userId) n.isRead = true; });
  res.json({ success: true });
});

app.delete('/api/notifications/item/:id', (req, res) => {
  const idx = notificationsData.findIndex(n => n.id === req.params.id);
  if (idx !== -1) notificationsData.splice(idx, 1);
  res.json({ success: true });
});

        function initHeroSlider() {
            const slide1 = document.getElementById('hero-slide-1');
            const slide2 = document.getElementById('hero-slide-2');
            if (!slide1 || !slide2) return;

            let currentSlide = 1;

            setInterval(() => {
                if (currentSlide === 1) {
                    slide1.classList.remove('active-slide');
                    slide1.classList.add('hidden-slide');

                    slide2.classList.remove('hidden-slide');
                    slide2.classList.add('active-slide');
                    currentSlide = 2;
                } else {
                    slide2.classList.remove('active-slide');
                    slide2.classList.add('hidden-slide');

                    slide1.classList.remove('hidden-slide');
                    slide1.classList.add('active-slide');
                    currentSlide = 1;
                }
            }, 7000);
        }

        async function updateHeaderUI() {
            const userArea = document.getElementById('header-user-area');
            const user = JSON.parse(localStorage.getItem('sw_user') || 'null');

            if (!user) {
                userArea.innerHTML = `<a href="/login.html" class="text-xs font-bold px-3.5 py-2 rounded-xl bg-[#3182f6] text-white hover:bg-[#1b64da] transition-all shadow-sm">로그인/가입</a>`;
                return;
            }

            const usersRes = await fetch('/api/users').then(r => r.json());
            const freshUser = (usersRes.users || []).find(u => u.id === user.id) || user;
            localStorage.setItem('sw_user', JSON.stringify(freshUser));

            const notiRes = await fetch(`/api/notifications/${user.id}`).then(r => r.json());
            const hasUnread = notiRes.hasUnread;

            let html = `
                <a href="/notifications.html" class="relative p-1.5 text-slate-500 hover:text-[#3182f6] transition-colors" title="알림 센터">
                    <svg class="w-6 h-6" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round">
                        <path d="M18 8A6 6 0 0 0 6 8c0 7-3 9-3 9h18s-3-2-3-9"></path>
                        <path d="M13.73 21a2 2 0 0 1-3.46 0"></path>
                    </svg>
                    ${hasUnread ? `<span class="absolute top-1 right-1 w-2.5 h-2.5 bg-[#3182f6] rounded-full ring-2 ring-white"></span>` : ''}
                </a>
            `;

            if (freshUser.role === 'senior') {
                html += `
                    <a href="/profile.html" class="text-xs font-extrabold bg-blue-50 text-[#3182f6] px-2.5 py-1.5 rounded-xl border border-blue-100 hover:bg-blue-100 transition-colors hidden sm:inline-block">
                        ${(freshUser.points || 0).toLocaleString()}P
                    </a>
                `;
            }

            html += `
                <a href="/profile.html" class="w-8 h-8 rounded-full bg-blue-50 text-[#3182f6] flex items-center justify-center font-bold border border-blue-200 hover:border-[#3182f6] transition-colors" title="프로필 페이지">
                    <i class="fa-solid fa-user text-xs"></i>
                </a>
                <button onclick="handleLogout()" class="text-xs font-bold text-slate-500 hover:text-rose-500 px-1.5 py-1 rounded-lg transition-colors">로그아웃</button>
            `;
            userArea.innerHTML = html;
        }

        async function applyJob() {
            const user = JSON.parse(localStorage.getItem('sw_user') || 'null');
            if (!user) {
                alert('로그인이 필요한 서비스입니다.');
                location.href = '/login.html';
                return;
            }
            if (user.role !== 'senior') {
                alert('시니어 구직자 회원만 지원할 수 있습니다.');
                return;
            }

            const res = await fetch(`/api/jobs/${currentJob.id}/apply`, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({
                    seniorId: user.id,
                    name: user.name,
                    phone: user.phone || '010-0000-0000'
                })
            }).then(r => r.json());

            if (res.success) {
                alert('지원이 완료되었습니다! 지원 현황 및 채용 상태는 내 프로필 페이지에서 확인하실 수 있습니다.');
                location.href = '/profile.html';
            } else {
                alert(res.message);
            }
        }

        async function updateHeaderUI() {
            const userArea = document.getElementById('header-user-area');
            const user = JSON.parse(localStorage.getItem('sw_user') || 'null');

            if (!user) {
                userArea.innerHTML = `<a href="/login.html" class="text-xs font-bold px-4 py-2 rounded-xl btn-toss shadow-sm">로그인 및 회원가입</a>`;
                return;
            }

            const usersRes = await fetch('/api/users').then(r => r.json());
            const freshUser = (usersRes.users || []).find(u => u.id === user.id) || user;
            localStorage.setItem('sw_user', JSON.stringify(freshUser));

            const notiRes = await fetch(`/api/notifications/${user.id}`).then(r => r.json());
            const hasUnread = notiRes.hasUnread;

            let html = `
                <a href="/notifications.html" class="relative p-2 text-slate-500 hover:text-[#3182f6] transition" title="알림 센터">
                    <svg class="w-6 h-6" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round">
                        <path d="M18 8A6 6 0 0 0 6 8c0 7-3 9-3 9h18s-3-2-3-9"></path>
                        <path d="M13.73 21a2 2 0 0 1-3.46 0"></path>
                    </svg>
                    ${hasUnread ? `<span class="absolute top-1 right-1 w-2.5 h-2.5 bg-[#3182f6] rounded-full ring-2 ring-white"></span>` : ''}
                </a>
            `;

            if (freshUser.role === 'senior') {
                html += `
                    <a href="/profile.html" class="text-xs font-extrabold bg-blue-50 text-[#3182f6] px-3 py-1.5 rounded-xl border border-blue-100 hover:bg-blue-100 transition">
                        포인트 ${(freshUser.points || 0).toLocaleString()}P (정산)
                    </a>
                `;
            }

            html += `
                <a href="/profile.html" class="w-9 h-9 rounded-full bg-blue-50 text-[#3182f6] flex items-center justify-center font-bold border border-blue-200 hover:border-[#3182f6] transition" title="프로필 페이지">

            `;
            userArea.innerHTML = html;
        }

            localStorage.removeItem('sw_user');

        function countJobs(mainRegion, subRegion = "전체") {
            return allServerJobs.filter(j => {
                const mainMatch = (j.regionMain === mainRegion) || 
                                 (shortToFull[j.regionMain] === mainRegion) ||
                                 (j.title && j.title.includes(mainRegion.substring(0, 2))) ||
                                 (j.storeName && j.storeName.includes(mainRegion.substring(0, 2)));

                if (!mainMatch) return false;
                if (subRegion === "전체") return true;

                return (j.regionSub === subRegion) || 
                       (j.title && j.title.includes(subRegion)) || 
                       (j.storeName && j.storeName.includes(subRegion));
            }).length;
        }

        function applyFiltersAndRender() {
            const keyword = document.getElementById('searchInput').value.trim().toLowerCase();
            const feedCard = document.getElementById('jobsFeedCard');

            const filtered = allServerJobs.filter(job => {
                let matchesMain = true;
                if (currentMainRegion !== "전체보기") {
                    const mainShort = currentMainRegion.substring(0, 2);
                    matchesMain = (job.regionMain === currentMainRegion) || 
                                 (shortToFull[job.regionMain] === currentMainRegion) ||
                                 (job.title && job.title.includes(mainShort)) ||
                                 (job.storeName && job.storeName.includes(mainShort));
                }

                let matchesSub = true;
                if (currentSubRegion !== "전체") {
                    matchesSub = (job.regionSub === currentSubRegion) ||
                                 (job.title && job.title.includes(currentSubRegion)) ||
                                 (job.storeName && job.storeName.includes(currentSubRegion));
                }

                let matchesSearch = true;
                if (keyword) {
                    matchesSearch = (job.title && job.title.toLowerCase().includes(keyword)) ||
                                    (job.storeName && job.storeName.toLowerCase().includes(keyword)) ||
                                    (job.category && job.category.toLowerCase().includes(keyword)) ||
                                    (job.address && job.address.toLowerCase().includes(keyword));
                }

                return matchesMain && matchesSub && matchesSearch;
            });

            if (filtered.length === 0) {
                const targetText = currentSubRegion !== "전체" ? `${currentMainRegion} ${currentSubRegion}` : currentMainRegion;
                feedCard.innerHTML = `
                    <div class="glass-card rounded-3xl p-16 text-center flex flex-col items-center justify-center space-y-3">
                        <div class="w-14 h-14 rounded-full bg-blue-100 text-[#3182f6] flex items-center justify-center text-2xl font-black shadow-inner mb-1">
                            <i class="fa-solid fa-exclamation"></i>
                        </div>
                        <p class="text-slate-500 font-bold text-sm">
                            '${targetText}' 지역에 등록된 구인 공고가 없습니다.
                        </p>
                    </div>
                `;
                return;
            }

            feedCard.innerHTML = filtered.map(job => `
                <div class="glass-card rounded-3xl p-6 md:p-7 flex flex-col sm:flex-row sm:items-center justify-between gap-5">
                    <div class="space-y-2">
                        <div class="flex items-center gap-2">
                            <span class="text-xs font-bold text-[#3182f6] bg-blue-50 px-2.5 py-1 rounded-lg border border-blue-100">${job.category || '음식'}</span>
                            <span class="text-xs font-bold text-slate-700 bg-slate-100 px-2.5 py-1 rounded-lg">${job.regionMain || '서울특별시'} ${job.regionSub || ''}</span>
                            ${job.isPeak ? '<span class="text-xs font-extrabold text-amber-600 bg-amber-50 px-2.5 py-1 rounded-lg border border-amber-200"><i class="fa-solid fa-thumbtack mr-1"></i>상단 고정</span>' : ''}
                            <span class="text-xs font-bold text-[#3182f6] bg-blue-50 px-2.5 py-1 rounded-lg border border-blue-100">${job.status}</span>
                        </div>
                        <h3 class="text-lg font-black text-slate-900 hover:text-[#3182f6] transition cursor-pointer" onclick="location.href='/job-detail.html?id=${job.id}'">
                            ${job.title}
                        </h3>
                        <p class="text-xs text-slate-500 font-medium">
                            <strong class="text-slate-800">${job.storeName}</strong> ${job.address ? `(${job.address})` : ''} | 근무일: ${job.date} (${job.time})
                        </p>
                    </div>
                    <div class="flex sm:flex-col items-end justify-between sm:justify-center gap-3 border-t sm:border-t-0 pt-3 sm:pt-0 border-slate-100">
                        <span class="text-lg font-black text-[#3182f6]">${parseInt(job.pay).toLocaleString()}원</span>
                        <a href="/job-detail.html?id=${job.id}" class="btn-toss text-xs font-bold px-5 py-2.5 rounded-xl shadow-sm whitespace-nowrap">
                            상세보기 / 지원
                        </a>
                    </div>
                </div>
            `).join('');
        }

        async function handleSubmit(e) {
            e.preventDefault();

                alert('비밀번호는 최소 8자리 이상이어야 합니다.');
                return;

                const name = document.getElementById('user-name').value || '회원';
                const phone = document.getElementById('user-phone').value || '';
                const privacyChecked = document.getElementById('privacy-check').checked;

                    return;

                const res = await fetch('/api/signup', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ id, pw, name, phone, role: currentRole })

                    alert(res.message);

                    method: 'POST',

        async function requestInvoiceConfirm(invoiceId) {
            await fetch('/api/invoices/request-confirm', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ invoiceId })
            });
            alert('관리자에게 수수료 입금 확인 요청이 전송되었습니다.');
            renderProfile();
        }

        async function requestCancelInvoice(invoiceId) {
            if (confirm('해당 채용 취소 건에 대한 명세서 취소 승인을 요청하시겠습니까?')) {
                await fetch('/api/invoices/request-cancel', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ invoiceId })
                });
                alert('명세서 취소 승인 요청이 접수되었습니다.');
                renderProfile();
            }
        }

        function openCancelModal(jobId, seniorId) {
            document.getElementById('cancel-job-id').value = jobId;
            document.getElementById('cancel-senior-id').value = seniorId;
            document.getElementById('cancel-agree-check').checked = false;
            document.getElementById('cancelHireModal').classList.remove('hidden');
        }
        function closeCancelModal() { document.getElementById('cancelHireModal').classList.add('hidden'); }

        async function confirmCancelHire() {
            const isAgreed = document.getElementById('cancel-agree-check').checked;
            if (!isAgreed) {
                alert('본사 접수 동의 체크박스에 동의해 주셔야 진행됩니다.');
                return;
            }

            const jobId = document.getElementById('cancel-job-id').value;
            const seniorId = document.getElementById('cancel-senior-id').value;

            await fetch(`/api/jobs/${jobId}/cancel-hire`, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ seniorId })
            }).then(r => r.json());

            closeCancelModal();
            alert('채용 취소 접수가 처리되었으며 다음 순위 지원자에게 조건 없이 승계 처리되었습니다.');
            renderProfile();
        }

        function openPayModal(jobId, seniorId, amount) {
            document.getElementById('pay-job-id').value = jobId;
            document.getElementById('pay-senior-id').value = seniorId;
            document.getElementById('pay-amount').value = amount;
            document.getElementById('pay-agree-check').checked = false;
            document.getElementById('payPointModal').classList.remove('hidden');
        }

        async function confirmPayPoint() {
            const isAgreed = document.getElementById('pay-agree-check').checked;
            if (!isAgreed) {
                alert('근무 확인 동의란을 체크해 주셔야 포인트 지급이 가능합니다.');
                return;
            }

            const jobId = document.getElementById('pay-job-id').value;
            const seniorId = document.getElementById('pay-senior-id').value;
            const basePay = document.getElementById('pay-amount').value;

            await fetch(`/api/jobs/${jobId}/pay-points`, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ seniorId, basePay })
            });

            closePayModal();
            alert('포인트 지급이 완료되었으며 수수료 납부 명세서가 정상 발행되었습니다.');
            renderProfile();
        }

        async function assignRank(jobId, seniorId, rank) {

            await fetch(`/api/jobs/${jobId}/rank`, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ seniorId, rank })
            });
            alert(`${rank} 지정 완료 및 시니어에게 알림이 전송되었습니다.`);
            renderProfile();
        }

            renderProfile();
        }

            const user = JSON.parse(localStorage.getItem('sw_user') || '{}');
            const res = await fetch('/api/chat-inquiries').then(r => r.json());
            const chats = (res.chats || []).filter(c => c.userId === user.id);
            const box = document.getElementById('chat-history-box');

            if (chats.length === 0) {
                box.innerHTML = `<p class="text-slate-400 text-center py-8">접수된 1:1 상담 문의 내역이 없습니다. 메시지를 남겨주세요.</p>`;
                return;
            }

            box.innerHTML = chats.map(c => `
                <div class="p-3 rounded-xl ${c.sender === 'admin' ? 'bg-blue-100/70 border border-blue-200 text-blue-950' : 'bg-white border border-slate-200 text-slate-800'} space-y-0.5">
                    <div class="flex justify-between items-center text-[10px]">
                        <strong class="font-bold">${c.sender === 'admin' ? '실버웍스 상담원' : '나의 문의'}</strong>
                        <span class="text-slate-400">${c.timestamp}</span>
                    </div>
                    <p class="font-medium">${c.message}</p>
                </div>
            `).join('');
            box.scrollTop = box.scrollHeight;
        }

        async function sendChatMessage() {
            const user = JSON.parse(localStorage.getItem('sw_user') || '{}');
            const msgInput = document.getElementById('chat-input');
            const message = msgInput.value.trim();
            if (!message) return;

            await fetch('/api/chat-inquiries', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ userId: user.id, userName: user.name, sender: 'user', message })
            });

            msgInput.value = '';
            loadChatHistory();
        }

        function renderStep() {
            const item = questions[currentStep];
            const container = document.getElementById('wizard-content');
            
            const total = 13;
            const pct = ((currentStep + 1) / total) * 100;
            document.getElementById('step-indicator').innerText = `${currentStep + 1} / ${total}`;
            document.getElementById('progress-bar').style.width = `${pct}%`;

            let html = `<label class="block font-bold text-slate-900 text-sm mb-3">${item.q}</label>`;

            if (item.type === "radio") {
                html += `<div class="space-y-2">`;
                item.options.forEach((opt) => {
                    const isSelected = answers[item.id] === opt;
                    html += `
                        <div onclick="selectRadio('${item.id}', '${opt}')" class="p-3.5 rounded-xl border border-slate-200 text-xs font-medium cursor-pointer hover:border-[#3182f6] transition ${isSelected ? 'border-[#3182f6] bg-blue-50 text-[#3182f6] font-bold' : 'bg-white text-slate-700'}">
                            ${opt}
                        </div>
                    `;
                });
                html += `</div>`;
                document.getElementById('next-btn').classList.add('hidden');
            } else if (item.type === "info") {
                html += `<p class="p-4 bg-blue-50 text-[#3182f6] rounded-xl font-bold text-xs">안내 사항입니다. 아래 다음 단계 버튼을 눌러 이동하세요.</p>`;
                document.getElementById('next-btn').classList.remove('hidden');
            } else {
                const val = answers[item.id] || '';
                html += `<input type="${item.type}" id="step-input" value="${val}" placeholder="${item.placeholder}" oninput="answers['${item.id}'] = this.value" class="w-full p-3.5 rounded-xl input-sky text-xs font-medium">`;
                document.getElementById('next-btn').classList.remove('hidden');
            }

            container.innerHTML = html;
            document.getElementById('prev-btn').classList.toggle('hidden', currentStep === 0);
            document.getElementById('next-btn').innerText = currentStep === questions.length - 1 ? "제출 완료" : "다음 단계";
        }

        function selectRadio(key, val) {
